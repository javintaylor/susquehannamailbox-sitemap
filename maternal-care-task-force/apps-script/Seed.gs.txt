/**
 * FILE: Seed.gs
 *
 * Seeds the content spreadsheet with the Task Force's starting content,
 * drawn from the 2026-09 Regional System of Care Action Team kickoff
 * (slides + meeting notes) and from the resource research compiled on
 * 2026-09-03 (see docs/RESEARCH_REPORT.txt).
 *
 * Idempotent per tab: a tab is seeded only if it has no data rows. To
 * re-seed a tab, clear its rows (keep the header) and run again.
 *
 * Contains NO patient information. Member rows contain professional
 * identity only (name, org, role) as stated on the record at a public
 * task-force meeting; contact columns are left blank and ShareContact is
 * FALSE until each member opts in.
 *
 * MEMBER CONSENT: these rows name real people and are transcribed from what
 * each person said aloud at the kickoff. Credentials and titles are as
 * spoken, not verified against a licence record. Every member should confirm
 * their own row before the site is shared, and any member who prefers not to
 * be listed should have Published set to FALSE.
 *
 * NEUTRALITY: Member listings describe a person's professional role only.
 * The site does not describe, rank, or promote any member organization's
 * services, and no resource row is added because a member requested it
 * without the same verification applied to every other row.
 *
 * Resource rows are in SeedResources.gs (separate file for readability).
 */

function seedAll() {
  requireAdmin_('SEED_ALL');
  seedPurpose();
  seedProblems();
  seedStructure();
  seedMembers();
  seedFunding();
  seedData();
  seedMeetings();
  seedActionItems();
  seedAnnouncements();
  seedResources(); // SeedResources.gs
  purgeCacheInternal_();
  console.log('Seeding complete.');
}

function purgeCacheInternal_() {
  CacheService.getScriptCache().remove('siteContent:v' + CONFIG.VERSION);
}

/** Appends rows only when the tab has no data rows. */
function seedIfEmpty_(tabName, rows) {
  var existing = readTable_(tabName);
  if (existing.length) { console.log(tabName + ': already has ' + existing.length + ' rows, skipped.'); return; }
  rows.forEach(function (r) { appendRow_(tabName, r); });
  console.log(tabName + ': seeded ' + rows.length + ' rows.');
}
