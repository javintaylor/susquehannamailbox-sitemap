 * Web app entry points. Deploy as: Execute as ME (script owner), Who has
 * access: "Anyone" (decision 2026-09-03: members use mixed Google and
 * non-Google identities, so the embed must load without sign-in). The
 * deployment URL is therefore public; the Published gate is the only
 * content control. Embed the URL in Google Sites via Insert > Embed > By URL.
 *
 * Every entry point: (1) authorization/identification, (2) audit write,
 * (3) work. In that order.
 *
 * All content returned is READ-ONLY and filtered to Published = TRUE rows.
 * No write endpoint exists in the web app by design (see Auth.gs).
 */

/** View names the site can be deep-linked to. Anything else is 'invalid'. */
var VALID_PAGES_ = ['home', 'problems', 'structure', 'funding', 'resources',
                    'data', 'meetings', 'barriers'];

/**
 * Serves the single-page member site.
 * @param {GoogleAppsScript.Events.DoGet} e
 * @return {GoogleAppsScript.HTML.HtmlOutput}
 */
function doGet(e) {
  // The page parameter is anonymous, attacker-controlled input that ends up
  // in the audit spreadsheet. Allowlist it at the entry point; anything that
  // is not a known view name is recorded as 'invalid', never echoed.
  var page = String((e && e.parameter && e.parameter.page) || 'home');
  if (VALID_PAGES_.indexOf(page) < 0) page = 'invalid';
  authorizeRead_('SitePage', page);
  var tpl = HtmlService.createTemplateFromFile('Index');
  tpl.appName = CONFIG.APP_NAME;
  tpl.version = CONFIG.VERSION;
  return tpl.evaluate()
    .setTitle(CONFIG.APP_NAME)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) // required for Google Sites embed
    .setSandboxMode(HtmlService.SandboxMode.IFRAME);
}

/**
 * Template helper: inlines another HTML file (Styles.html, Client.html).
 * @param {string} filename
 * @return {string}
 */
function include(filename) {
  return HtmlService.createHtmlOutputFromFile(filename).getContent();
}

/**
 * Returns every published content block the site needs, in one call.
 * Cached for CONFIG.CACHE_TTL_SECONDS to keep Sheets reads low.
 * @return {Object}
 */
function getSiteContent() {
  var who = authorizeRead_('SiteContent', 'all');
  var cache = CacheService.getScriptCache();
  var cached = cache.get('siteContent:v' + CONFIG.VERSION);
  if (cached) {
    var parsed = JSON.parse(cached);
    parsed.viewerRole = who.role;
    return parsed;
  }
  var bundle = buildSiteContent_();
  bundle.viewerRole = who.role;
  try {
    var json = JSON.stringify(bundle);
    if (json.length < 100000) cache.put('siteContent:v' + CONFIG.VERSION, json, CONFIG.CACHE_TTL_SECONDS);
  } catch (err) {
    console.warn('CACHE_PUT_SKIPPED');
  }
  return bundle;
}

/** The only Config keys ever sent to a browser. */
var SERVED_CONFIG_KEYS_ = ['SITE_TITLE', 'TAGLINE', 'NEXT_MEETING'];

/**
 * Assembles the published content bundle from the sheet. Internal.
 * @return {Object}
 */
function buildSiteContent_() {
  var T = CONFIG.TABS;
  var configAll = readConfigMap_();
  // The Config tab has NO Published column and cannot have one, so it can
  // never be gated. Serve only the keys the UI actually renders; everything
  // else (form IDs, notes, future operational keys) stays server-side.
  var config = SERVED_CONFIG_KEYS_.reduce(function (o, k) {
    o[k] = configAll[k] || '';
    return o;
  }, {});
  var members = readPublished_(T.MEMBERS).map(redactMemberContact_);
  var today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  var announcements = readPublished_(T.ANNOUNCEMENTS).filter(function (a) {
    return !a.ExpiresOn || String(a.ExpiresOn) >= today;
  });
  return {
    generatedAt: new Date().toISOString(),
    appName: CONFIG.APP_NAME,
    version: CONFIG.VERSION,
    config: config,
    purpose: sortByOrder_(readPublished_(T.PURPOSE)),
    problems: sortByOrder_(readPublished_(T.PROBLEMS)),
    structure: sortByOrder_(readPublished_(T.STRUCTURE)),
    members: sortByOrder_(members),
    funding: sortByOrder_(readPublished_(T.FUNDING)),
    resources: sortByOrder_(readPublished_(T.RESOURCES)),
    data: sortByOrder_(readPublished_(T.DATA)),
    meetings: readPublished_(T.MEETINGS).sort(function (a, b) { return String(b.Date).localeCompare(String(a.Date)); }),
    actionItems: readPublished_(T.ACTION_ITEMS),
    announcements: announcements.sort(function (a, b) { return String(b.Date).localeCompare(String(a.Date)); }),
    barriers: buildBarrierSummary_(),
    forms: {
      barrierFormUrl: configAll.BARRIER_FORM_URL || '',
      resourceFormUrl: configAll.RESOURCE_FORM_URL || ''
    }
  };
}

/**
 * Strips Email/Phone unless the member has consented (ShareContact = TRUE).
 * Internal notes never leave the sheet.
 * @param {Object} m
 * @return {Object}
 */
function redactMemberContact_(m) {
  var out = {};
  Object.keys(m).forEach(function (k) {
    if (k === 'InternalNotes') return;
    out[k] = m[k];
  });
  if (!isTruthyFlag_(m[CONFIG.MEMBER_CONTACT_CONSENT_COLUMN])) {
    out.Email = '';
    out.Phone = '';
  }
  return out;
}

/** Closed vocabularies mirroring the Form's dropdown choices (Setup.gs). */
var BARRIER_COUNTIES_ = ['Cecil', 'Harford', 'Both / Regional', 'Other'];
var BARRIER_CATEGORIES_ = [
  'Transportation cost or availability', 'Emergency / EMS transport',
  'Insurance not accepted (incl. Maryland Medicaid out of state)',
  'No in-county provider for service', 'Appointment availability / scheduling',
  'Maternal-fetal medicine access', 'Behavioral health / perinatal mental health',
  'Substance use treatment access', 'Material needs (car seat, crib, supplies)',
  'Language / communication', 'Housing instability',
  'Childcare to attend appointments', 'Other'];
var BARRIER_SERVICES_ = [
  'Prenatal care', 'Labor and delivery', 'Induction / scheduled procedure',
  'Postpartum care', 'Newborn / pediatric', 'Lactation', 'Behavioral health',
  'WIC / nutrition', 'Home visiting', 'Other'];

/**
 * Returns the value only when it is in the allowed list, else 'Unspecified'.
 * @param {*} v
 * @param {string[]} allowed
 * @return {string}
 */
function oneOf_(v, allowed) {
  var s = String(v === undefined || v === null ? '' : v).trim();
  return allowed.indexOf(s) >= 0 ? s : 'Unspecified';
}

/**
 * Accepts only YYYY-MM. Anything else counts toward the total but
 * contributes no text to the page.
 * @param {*} v
 * @return {string}
 */
function normalizeMonth_(v) {
  var m = String(v === undefined || v === null ? '' : v).match(/^(\d{4})-(\d{2})/);
  if (!m) return 'Unspecified';
  var mo = parseInt(m[2], 10);
  if (mo < 1 || mo > 12) return 'Unspecified';
  return m[1] + '-' + m[2];
}

/**
 * Aggregates the de-identified BarrierLog into counts. Free-text descriptions
 * are exposed ONLY for rows an admin has marked Published = TRUE after review.
 * Reporter email addresses and reporting organizations are never returned.
 * @return {Object}
 */
function buildBarrierSummary_() {
  var rows;
  try { rows = readTable_(CONFIG.TABS.BARRIER_LOG); } catch (e) { return { total: 0, byCategory: {}, byCounty: {}, byMonth: {}, published: [] }; }
  var byCategory = {}, byCounty = {}, byMonth = {}, published = [];
  rows.forEach(function (r) {
    // Closed-list fields come from Form dropdowns, but never trust that a
    // header still maps to a dropdown — anything unrecognized becomes a
    // fixed bucket label so unreviewed free text can never surface as a
    // chart label, bypassing co-chair review.
    var cat = oneOf_(r['Barrier category'], BARRIER_CATEGORIES_);
    var county = oneOf_(r['County'], BARRIER_COUNTIES_);
    var month = normalizeMonth_(r['Month of occurrence']);
    byCategory[cat] = (byCategory[cat] || 0) + 1;
    byCounty[county] = (byCounty[county] || 0) + 1;
    byMonth[month] = (byMonth[month] || 0) + 1;
    if (isTruthyFlag_(r[CONFIG.PUBLISHED_COLUMN])) {
      // The reporting organization is deliberately NOT published. In a
      // two-county region a single-site program plus a month plus a service
      // type can re-identify one family; the org adds nothing the Task Force
      // cannot get from the sheet itself.
      published.push({
        month: month, county: county, category: cat,
        service: oneOf_(r['Service type'], BARRIER_SERVICES_),
        description: String(r['Brief description'] || '').slice(0, CONFIG.BARRIER_TEXT_MAX)
      });
    }
  });
  return { total: rows.length, byCategory: byCategory, byCounty: byCounty, byMonth: byMonth, published: published };
}

/**
 * ADMIN (run from editor): clears the content cache so sheet edits show
 * immediately instead of after CONFIG.CACHE_TTL_SECONDS.
 */
function purgeCache() {
  requireAdmin_('PURGE_CACHE');
  CacheService.getScriptCache().remove('siteContent:v' + CONFIG.VERSION);
}

/**
 * ADMIN (run from editor): smoke test that prints row counts per tab.
 * Prints counts only, never row contents.
 */
function smokeTest() {
  requireAdmin_('SMOKE_TEST');
  var b = buildSiteContent_();
  var summary = ['purpose', 'problems', 'structure', 'members', 'funding', 'resources', 'data', 'meetings', 'actionItems', 'announcements']
    .map(function (k) { return k + '=' + b[k].length; }).join(', ');
  console.log('Published row counts: ' + summary + ', barriers=' + b.barriers.total);
}
