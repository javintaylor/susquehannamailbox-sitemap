/**
 * FILE: Audit.gs
 *
 * Audit writes go to a SEPARATE spreadsheet in a Drive location no site
 * user can reach. The site holds no PHI, but the same discipline applies:
 * every entry-point call is logged with actor, action, resource, outcome.
 * Entries never contain content — only that something was read and what.
 *
 * Retention: keep the audit spreadsheet a minimum of six years
 * (45 CFR 164.316(b)(2)(i)) even though no PHI is expected; the log is the
 * evidence that none was handled.
 */

/**
 * Makes one value safe to store in a spreadsheet cell.
 *
 * appendRow uses setValues semantics, so a string beginning with = + - @
 * (or a leading tab/CR) is stored as a live FORMULA. Audit fields can carry
 * attacker-controlled text (an anonymous URL parameter reaches resourceId),
 * so every field is coerced to a string, prefixed with an apostrophe when it
 * could be parsed as a formula, and truncated. An audit log must never
 * execute what it records.
 *
 * @param {*} v
 * @param {number} [max]
 * @return {string}
 */
function auditSafe_(v, max) {
  var s = (v === undefined || v === null) ? '' : String(v);
  s = s.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, ' ');
  if (/^[=+\-@\t\r]/.test(s)) s = "'" + s;
  var limit = max || 300;
  return s.length > limit ? s.slice(0, limit) + '…' : s;
}

/**
 * Entries that could not be written because the audit spreadsheet did not
 * exist yet (only possible during the very first runSetup, before step 2
 * creates it). Held in memory for the current execution and replayed by
 * flushPendingAudit_ as soon as the audit spreadsheet exists, so the
 * bootstrap action still lands in the log.
 * @type {Object[]}
 */
var PENDING_AUDIT_ = [];

/**
 * Writes one audit row. Never throws to the caller — an audit failure is
 * logged to Stackdriver (without content) and the request is still refused
 * upstream if the audit could not be written before a write action.
 *
 * @param {Object} entry
 * @param {string} entry.actor        email or 'anonymous-google-account' / 'anonymous'
 * @param {string} entry.actorRole    'admin' | 'member' | 'public'
 * @param {string} entry.action       e.g. 'READ', 'SUBMIT', 'SETUP', 'DENY'
 * @param {string} entry.resourceType e.g. 'SiteContent', 'BarrierLog'
 * @param {string} [entry.resourceId]
 * @param {string} entry.outcome      'SUCCESS' | 'DENIED' | 'ERROR'
 * @param {string} [entry.session]
 * @param {string} [entry.justification]
 * @return {boolean} true if written
 */
function writeAudit_(entry) {
  // Bootstrap case: the audit spreadsheet is created partway through the
  // first runSetup, so the SETUP entry that precedes it has nowhere to go.
  // Buffer it instead of dropping it; flushPendingAudit_ replays it.
  if (!PropertiesService.getScriptProperties().getProperty(CONFIG.PROP_AUDIT_SS_ID)) {
    PENDING_AUDIT_.push({ entry: entry, occurredAt: new Date() });
    console.log('AUDIT_DEFERRED action=' + (entry && entry.action) +
                ' (audit spreadsheet not provisioned yet; will be replayed)');
    return false;
  }
  try {
    var ss = SpreadsheetApp.openById(getAuditSpreadsheetId_());
    var sheet = ss.getSheetByName(CONFIG.AUDIT_TAB) || ss.insertSheet(CONFIG.AUDIT_TAB);
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['TimestampUTC', 'LocalTZ', 'Actor', 'ActorRole', 'Action',
        'ResourceType', 'ResourceId', 'Outcome', 'Session', 'Justification', 'AppVersion']);
      sheet.setFrozenRows(1);
    }
    var now = new Date();
    sheet.appendRow([
      Utilities.formatDate(now, 'UTC', "yyyy-MM-dd'T'HH:mm:ss'Z'"),
      Session.getScriptTimeZone(),
      auditSafe_(entry.actor || 'unknown', 254),
      auditSafe_(entry.actorRole || 'unknown', 40),
      auditSafe_(entry.action || 'UNKNOWN', 40),
      auditSafe_(entry.resourceType || '', 60),
      auditSafe_(entry.resourceId || '', 200),
      auditSafe_(entry.outcome || 'UNKNOWN', 20),
      auditSafe_(entry.session || '', 40),
      auditSafe_(entry.justification || '', 500),
      CONFIG.VERSION
    ]);
    return true;
  } catch (e) {
    // No content, no identifiers beyond the action name, in the console log.
    console.error('AUDIT_WRITE_FAILED action=' + (entry && entry.action));
    return false;
  }
}

/**
 * Replays any entries buffered before the audit spreadsheet existed.
 * Called by runSetup immediately after the audit spreadsheet is created.
 * Each replayed row keeps its ORIGINAL timestamp and carries a
 * justification saying it was written after the fact, so the log is
 * complete without misrepresenting when the action happened.
 * @return {number} how many rows were replayed
 */
function flushPendingAudit_() {
  if (!PENDING_AUDIT_.length) return 0;
  var buffered = PENDING_AUDIT_.slice();
  PENDING_AUDIT_ = [];
  var written = 0;
  buffered.forEach(function (item) {
    var e = item.entry || {};
    var note = 'Deferred write: audit spreadsheet did not exist at time of ' +
               'action (bootstrap). Original UTC timestamp ' +
               Utilities.formatDate(item.occurredAt, 'UTC', "yyyy-MM-dd'T'HH:mm:ss'Z'") + '.';
    var ok = writeAudit_({
      actor: e.actor, actorRole: e.actorRole, action: e.action,
      resourceType: e.resourceType, resourceId: e.resourceId,
      outcome: e.outcome, session: e.session,
      justification: e.justification ? e.justification + ' | ' + note : note
    });
    if (ok) written++;
  });
  return written;
}

/**
 * Returns an opaque per-execution session token for correlating audit rows.
 * Not derived from any user attribute.
 * @return {string}
 */
function newSessionToken_() {
  return Utilities.getUuid().slice(0, 8);
}
