<script>
(function () {
  'use strict';

  var STATE = { content: null, view: 'home' };
  var $ = function (sel, root) { return (root || document).querySelector(sel); };
  var $$ = function (sel, root) { return Array.prototype.slice.call((root || document).querySelectorAll(sel)); };

  /* ---------- safe HTML helpers (all sheet text is escaped) ---------- */
  function esc(s) {
    return String(s === undefined || s === null ? '' : s)
      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
  }
  // Every newline starts a new paragraph. A forced <br> mid-sentence reads
  // as a broken line, so this never emits one.
  function para(s) {
    return String(s || '').split(/\r?\n+/).map(function (t) { return t.trim(); })
      .filter(Boolean).map(function (t) { return '<p>' + esc(t) + '</p>'; }).join('');
  }
  // Fields that hold one item per line (decisions, next steps) render as a
  // real list rather than run-on text split by line breaks.
  function bullets(s) {
    var items = String(s || '').split(/\r?\n+/).map(function (t) { return t.trim(); }).filter(Boolean);
    if (!items.length) return '';
    if (items.length === 1) return '<p>' + esc(items[0]) + '</p>';
    return '<ul>' + items.map(function (t) { return '<li>' + esc(t) + '</li>'; }).join('') + '</ul>';
  }
  function isUrl(u) { return /^https?:\/\//i.test(String(u || '')); }
  function link(url, label) {
    if (!isUrl(url)) return esc(label || url || '');
    return '<a href="' + esc(url) + '" target="_blank" rel="noopener">' + esc(label || 'Website') + '</a>';
  }
  function truncate(s, n) {
    var t = String(s || '').replace(/\s+/g, ' ').trim();
    return t.length > n ? t.slice(0, n - 1).replace(/[\s,;:.]+\S*$/, '') + '…' : t;
  }

  /* ---------- disclosure ---------- */
  // Drawn icon, one stroke weight, rotated by CSS on open.
  var CHEV = '<svg class="chev" viewBox="0 0 16 16" fill="none" aria-hidden="true">' +
    '<path d="M6 3.5L10.5 8L6 12.5" stroke="currentColor" stroke-width="1.6" stroke-linecap="round" stroke-linejoin="round"/></svg>';

  /**
   * One collapsible block. `sum` is {title, meta, tags}; `body` is HTML.
   * Native <details> so keyboard and screen-reader behaviour come for free.
   */
  function disc(sum, body, open) {
    return '<details class="disc"' + (open ? ' open' : '') + '><summary>' + CHEV +
      '<span class="sum"><span class="sum-title">' + sum.title + '</span>' +
      (sum.meta ? '<span class="sum-meta">' + sum.meta + '</span>' : '') +
      (sum.tags ? '<span class="sum-tags">' + sum.tags + '</span>' : '') +
      '</span></summary><div class="disc-body">' + body + '</div></details>';
  }
  /**
   * The strip under each page heading. `actions` are the things a member can
   * do here (open a form); `jumps` scroll to a section on this page. Rendered
   * only when there is something to put in it.
   * @param {Array<{href:string,label:string,primary?:boolean}>} actions
   * @param {Array<{id:string,label:string}>} jumps
   */
  function pageBar(actions, jumps) {
    var a = (actions || []).filter(function (x) { return x && x.href; });
    var j = (jumps || []).filter(Boolean);
    if (!a.length && !j.length) return '';
    var html = '<div class="pagebar">';
    if (a.length) {
      html += '<div class="pagebar-actions">' + a.map(function (x) {
        return '<a class="btn' + (x.primary ? '' : ' secondary') + '" target="_blank" rel="noopener" href="' + esc(x.href) + '">' + esc(x.label) + '</a>';
      }).join('') + '</div>';
    }
    if (j.length) {
      html += '<nav class="pagebar-jumps" aria-label="On this page"><span class="pagebar-label">On this page</span>' +
        j.map(function (x) { return '<a href="#' + esc(x.id) + '" data-jump="' + esc(x.id) + '">' + esc(x.label) + '</a>'; }).join('') + '</nav>';
    }
    return html + '</div>';
  }

  function accBar(scopeId, countId) {
    return '<div class="acc-bar"><button type="button" data-acc="open" data-scope="' + scopeId + '">Expand all</button>' +
      '<button type="button" data-acc="close" data-scope="' + scopeId + '">Collapse all</button>' +
      (countId ? '<span class="count" id="' + countId + '"></span>' : '') + '</div>';
  }
  document.addEventListener('click', function (ev) {
    var b = ev.target.closest && ev.target.closest('[data-acc]');
    if (b) {
      var scope = document.getElementById(b.dataset.scope);
      if (scope) $$('details', scope).forEach(function (d) { d.open = b.dataset.acc === 'open'; });
      return;
    }
    var jump = ev.target.closest && ev.target.closest('[data-jump]');
    if (jump) {
      ev.preventDefault();
      var target = document.getElementById(jump.getAttribute('data-jump'));
      if (target) {
        var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
        var top = target.getBoundingClientRect().top + window.scrollY - stickyOffset() - 12;
        window.scrollTo({ top: top, behavior: reduce ? 'auto' : 'smooth' });
        target.setAttribute('tabindex', '-1');
        target.focus({ preventScroll: true });
      }
      return;
    }
    var g = ev.target.closest && ev.target.closest('[data-goto]');
    if (g) { ev.preventDefault(); show(g.getAttribute('data-goto')); }
  });

  // The tab bar is sticky, so a jump must stop short of it.
  function stickyOffset() {
    var t = $('.tabs');
    return t ? Math.round(t.getBoundingClientRect().height) : 0;
  }

  /* ---------- indicator tagging ---------- */
  var INDICATOR_ORDER = ['Infant mortality', 'Preterm birth', 'Low birth weight',
    'Very low birth weight', 'Early prenatal care', 'No prenatal care', 'Cross-cutting'];
  function splitInd(s) { return String(s || '').split(/[;,]/).map(function (x) { return x.trim(); }).filter(Boolean); }
  function indicatorPills(s) {
    return splitInd(s).map(function (x) { return '<span class="pill ind-tag">' + esc(x) + '</span>'; }).join('');
  }
  function indicatorSelect(id, rows, field) {
    var present = uniq(rows.reduce(function (acc, r) { return acc.concat(splitInd(r[field])); }, []))
      .sort(function (a, b) { var ia = INDICATOR_ORDER.indexOf(a), ib = INDICATOR_ORDER.indexOf(b); return (ia < 0 ? 99 : ia) - (ib < 0 ? 99 : ib); });
    return '<label>Indicator <select id="' + id + '"><option value="">All</option>' +
      present.map(function (k) { return '<option>' + esc(k) + '</option>'; }).join('') + '</select></label>';
  }
  function hasInd(row, field, ind) { return !ind || splitInd(row[field]).indexOf(ind) >= 0; }

  function countyPill(c) {
    var s = String(c || '').toLowerCase();
    var cls = (s.indexOf('both') >= 0 || s.indexOf('regional') >= 0 || (s.indexOf('cecil') >= 0 && s.indexOf('harford') >= 0)) ? 'both'
      : s.indexOf('cecil') >= 0 ? 'cecil' : s.indexOf('harford') >= 0 ? 'harford' : '';
    return c ? '<span class="pill ' + cls + '">' + esc(c) + '</span>' : '';
  }
  function statusPill(s) {
    if (!s) return '';
    return '<span class="pill status-' + esc(String(s).toLowerCase().replace(/[^a-z]/g, '')) + '">' + esc(s) + '</span>';
  }

  /* ---------- navigation ---------- */
  function show(view) {
    STATE.view = view;
    $$('.view').forEach(function (v) { v.hidden = v.id !== 'view-' + view; });
    $$('.tab').forEach(function (t) {
      if (t.dataset.view === view) t.setAttribute('aria-current', 'page'); else t.removeAttribute('aria-current');
    });
    try { history.replaceState(null, '', '#' + view); } catch (e) { /* embedded */ }
    window.scrollTo({ top: 0 });
  }
  $$('.tab').forEach(function (t) { t.addEventListener('click', function () { show(t.dataset.view); }); });

  /* ---------- home ---------- */
  function renderHome(c) {
    var cfg = c.config || {};
    $('#siteTitle').textContent = cfg.SITE_TITLE || 'Maternal Care Task Force';
    $('#siteSubtitle').textContent = cfg.SUBTITLE || 'Cecil and Harford Counties, Maryland';
    $('#siteTagline').textContent = cfg.TAGLINE || '';
    renderIndicatorStrip(c.data || []);

    var ann = c.announcements || [];
    var bar = $('#announcementBar');
    if (ann.length) {
      bar.hidden = false;
      bar.innerHTML = '<strong>' + esc(ann[0].Title) + '</strong> ' + esc(ann[0].Body) +
        (ann[0].LinkURL ? ' ' + link(ann[0].LinkURL, 'More') : '');
    }

    var sections = {};
    (c.purpose || []).forEach(function (p) {
      var k = p.Section || 'Purpose';
      (sections[k] = sections[k] || []).push(p);
    });
    var keys = Object.keys(sections);
    var jumps = keys.map(function (sec, i) { return { id: 'sec-purpose-' + i, label: sec }; });
    if (cfg.NEXT_MEETING) jumps.push({ id: 'sec-next-meeting', label: 'Next meeting' });
    var html = '<h2>Purpose</h2><p class="lead">Why the Task Force exists, what it is working to improve, and how it is organised.</p>' +
      pageBar([
        { href: (c.forms || {}).barrierFormUrl, label: 'Report a barrier', primary: true },
        { href: (c.forms || {}).resourceFormUrl, label: 'Suggest a resource' }
      ], jumps) +
      accBar('purposeList') + '<div id="purposeList">';
    keys.forEach(function (sec, i) {
      var body = sections[sec].map(function (p) {
        return (p.Heading ? '<h4>' + esc(p.Heading) + '</h4>' : '') + para(p.Body);
      }).join('');
      var n = sections[sec].length;
      html += '<div id="sec-purpose-' + i + '">' +
        disc({ title: esc(sec), meta: n > 1 ? n + ' sections' : '' }, body, i === 0) + '</div>';
    });
    html += '</div>';
    if (cfg.NEXT_MEETING) html += '<div class="section" id="sec-next-meeting"><div class="card"><h3>Next meeting</h3><p>' + esc(cfg.NEXT_MEETING) + '</p></div></div>';
    if (ann.length > 1) {
      html += '<div class="section"><h3>Announcements</h3>' + ann.map(function (a) {
        return '<div class="card"><p class="meta">' + esc(a.Date) + '</p><h4>' + esc(a.Title) + '</h4>' +
          para(a.Body) + (a.LinkURL ? '<p>' + link(a.LinkURL, 'Link') + '</p>' : '') + '</div>';
      }).join('') + '</div>';
    }
    $('#view-home').innerHTML = html;
  }

  // Six tiles: the outcomes the whole site is organised around.
  function renderIndicatorStrip(rows) {
    var strip = $('#indicatorStrip');
    if (!strip) return;
    if (!rows.length) { strip.hidden = true; return; }
    strip.hidden = false;
    var src = rows[0] || {};
    var note = $('#indicatorSource');
    if (note) {
      note.innerHTML = 'Cecil shown. ' + (src.Year ? esc(src.Year) + ', ' : '') +
        (src.SourceURL ? link(src.SourceURL, src.Source || 'source') : esc(src.Source || '')) +
        ' · <a href="#data" data-goto="data">All six indicators and sources</a>';
      note.hidden = false;
    }
    strip.innerHTML = rows.map(function (r) {
      var n = parseFloat(r.Cecil), m = parseFloat(r.Maryland);
      var higherBetter = /early prenatal|adequate|coverage|breastfeed/i.test(String(r.Indicator));
      var cls = (!isNaN(n) && !isNaN(m) && n !== m) ? ((higherBetter ? n > m : n < m) ? 'better' : 'worse') : '';
      var unit = r.Unit === '%' ? '%' : '';
      return '<button class="ind ' + cls + '" data-goto="data"><span class="n">' + esc(r.Indicator) + '</span>' +
        '<span class="v">' + esc(r.Cecil) + unit + '</span>' +
        '<span class="c">Harford ' + esc(r.Harford) + unit + ' · Maryland ' + esc(r.Maryland) + unit + '</span></button>';
    }).join('');
  }

  /* ---------- problems ---------- */
  function evidenceLine(p) {
    if (!p.SourceOrEvidence && !p.SourceURL) return '';
    return '<p class="meta">Evidence: ' +
      (p.SourceURL ? link(p.SourceURL, p.SourceOrEvidence || 'Source') : esc(p.SourceOrEvidence)) + '</p>';
  }
  function renderProblems(c) {
    var problems = c.problems || [];
    $('#view-problems').innerHTML = '<h2>The Problem</h2>' +
      '<p class="lead">Access concerns the Task Force is working to solve. Each is tied to an Action Team and to the indicators it is expected to move.</p>' +
      pageBar([{ href: (c.forms || {}).barrierFormUrl, label: 'Report a barrier', primary: true }],
        [{ id: 'sec-problem-filters', label: 'Filter' }, { id: 'problemList', label: 'All problems' }]) +
      '<div class="filters" id="sec-problem-filters">' + indicatorSelect('fProbInd', problems, 'TargetIndicators') + '</div>' +
      accBar('problemList', 'pCount') + '<div id="problemList"></div>';
    function apply() {
      var ind = $('#fProbInd').value;
      var list = problems.filter(function (p) { return hasInd(p, 'TargetIndicators', ind); });
      $('#pCount').textContent = list.length + ' of ' + problems.length;
      $('#problemList').innerHTML = list.map(function (p) {
        return disc({
          title: esc(p.Title),
          meta: [esc(p.Counties), esc(p.RelatedActionTeam)].filter(Boolean).join(' · '),
          tags: indicatorPills(p.TargetIndicators)
        }, para(p.Description) + evidenceLine(p));
      }).join('') || '<p class="muted">No problems match.</p>';
    }
    $('#fProbInd').addEventListener('change', apply);
    apply();
  }

  /* ---------- structure ---------- */
  function onTeam(memberTeams, teamName) {
    var norm = function (s) { return String(s || '').toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim(); };
    var target = norm(teamName);
    if (!target) return false;
    return String(memberTeams || '').split(/[;,]/).map(norm).filter(Boolean).some(function (t) {
      return t === target || target.indexOf(t) === 0 || t.indexOf(target) === 0;
    });
  }
  function memberList(list) {
    if (!list.length) return '<p class="muted small">No members listed yet.</p>';
    return '<div class="table-scroll"><table><thead><tr><th>Name</th><th>Role</th><th>Organisation</th><th>County</th></tr></thead><tbody>' +
      list.map(function (m) {
        var contact = '';
        if (m.Email) contact += '<div class="small"><a href="mailto:' + esc(m.Email) + '">' + esc(m.Email) + '</a></div>';
        if (m.Phone) contact += '<div class="small">' + esc(m.Phone) + '</div>';
        return '<tr><td><strong>' + esc(m.Name) + (m.Credentials ? ', ' + esc(m.Credentials) : '') + '</strong>' + contact + '</td>' +
          '<td>' + esc(m.RoleTitle) + '</td><td>' + esc(m.Organization) + '</td><td>' + esc(m.County) + '</td></tr>';
      }).join('') + '</tbody></table></div>';
  }
  function renderStructure(c) {
    var structure = c.structure || [], members = c.members || [];
    var steering = structure.filter(function (s) { return /steering/i.test(s.Level); });
    var teams = structure.filter(function (s) { return /action/i.test(s.Level); });

    function block(s, roster) {
      var body = para(s.Purpose) +
        (s.CoChairs ? '<p><strong>Co-chairs:</strong> ' + esc(s.CoChairs) + '</p>' : '') +
        (s.StaffSupport ? '<p><strong>Staff support:</strong> ' + esc(s.StaffSupport) + '</p>' : '') +
        (s.MeetingCadence ? '<p><strong>Cadence:</strong> ' + esc(s.MeetingCadence) + '</p>' : '') +
        (s.TargetIndicators ? '<p class="small"><strong>Indicators:</strong> ' + indicatorPills(s.TargetIndicators) + '</p>' : '') +
        '<h4 style="margin-top:14px">Members (' + roster.length + ')</h4>' + memberList(roster);
      return disc({
        title: esc(s.Name),
        meta: [statusPill(s.Status), roster.length + ' member' + (roster.length === 1 ? '' : 's'),
          esc(s.MeetingCadence || '')].filter(Boolean).join(' · ')
      }, body);
    }

    var html = '<h2>Structure</h2><p class="lead">A Steering Committee sets direction. Four Action Teams do the work. Only the Regional System of Care team has met, so the other rosters are empty by fact rather than by omission.</p>' +
      pageBar([], [{ id: 'structureList', label: 'Steering Committee' }, { id: 'sec-teams', label: 'Action Teams' }]) +
      accBar('structureList') + '<div id="structureList">';
    steering.forEach(function (s) { html += block(s, members.filter(function (m) { return truthy(m.SteeringCommittee); })); });
    html += '</div><div class="section" id="sec-teams"><h3>Action Teams</h3><div id="teamList">';
    teams.forEach(function (t) { html += block(t, members.filter(function (m) { return onTeam(m.ActionTeams, t.Name); })); });
    html += '</div><p class="notice small" style="margin-top:14px">Contact details appear only for members who opted in. Every listing states professional role only. To correct your entry, contact the co-chairs.</p></div>';
    $('#view-structure').innerHTML = html;
  }

  /* ---------- funding ---------- */
  function renderFunding(c) {
    var funding = c.funding || [];
    $('#view-funding').innerHTML = '<h2>Funding</h2>' +
      '<p class="lead">Grants and programs resourcing the Task Force and the regional system of care.</p>' +
      pageBar([], funding.map(function (f, i) { return { id: 'sec-fund-' + i, label: truncate(f.Name, 34) }; })) +
      accBar('fundingList') + '<div id="fundingList">' + funding.map(function (f, i) {
        return disc({
          title: esc(f.Name),
          meta: [esc(f.Source), esc(f.Amount)].filter(Boolean).join(' · '),
          tags: countyPill(f.Counties) + statusPill(f.Status) + indicatorPills(f.Indicators)
        }, para(f.Description) +
          (f.IntendedUse ? '<p><strong>Intended use:</strong> ' + esc(f.IntendedUse) + '</p>' : '') +
          (f.URL ? '<p>' + link(f.URL, 'Program page') + '</p>' : ''));
      }).map(function (h, i) { return '<div id="sec-fund-' + i + '">' + h + '</div>'; }).join('') + '</div>';
  }

  /* ---------- resources ---------- */
  function linksLine(r) {
    var parts = [];
    if (isUrl(r.URL)) parts.push(link(r.URL, 'Website'));
    if (isUrl(r.SourceURL) && r.SourceURL !== r.URL) parts.push(link(r.SourceURL, 'Source'));
    return parts.length ? '<p>' + parts.join(' · ') + '</p>' : '';
  }
  function provenanceLine(r) {
    var src = String(r.SourceURL || '');
    var memberReport = src && !isUrl(src);
    var bits = [];
    if (memberReport) bits.push('Reported by a Task Force member — not independently verified');
    else if (r.DateVerified) bits.push('Verified ' + esc(r.DateVerified));
    if (r.Confidence) bits.push('Confidence: ' + esc(r.Confidence));
    if (memberReport && src) bits.push(esc(src));
    return bits.length ? '<p class="meta">' + bits.join(' · ') + '</p>' : '';
  }
  function countyMatches(cellText, filter) {
    var wide = cellText.indexOf('both') >= 0 || cellText.indexOf('regional') >= 0;
    var statewide = cellText.indexOf('statewide') >= 0 || cellText.indexOf('national') >= 0;
    if (filter === 'cecil' || filter === 'harford') return cellText.indexOf(filter) >= 0 || wide || statewide;
    if (filter === 'both') return wide || statewide;
    if (filter === 'statewide') return statewide;
    return true;
  }
  function renderResources(c) {
    var all = c.resources || [];
    var cats = uniq(all.map(function (r) { return r.Category; }));
    var types = uniq(all.map(function (r) { return r.Type; }));
    $('#view-resources').innerHTML = '<h2>Resources &amp; Programs</h2>' +
      '<p class="lead">Federal, state, and local programs for perinatal care in Cecil and Harford Counties, grouped by what they do. Confirm details with the provider before referring.</p>' +
      pageBar([{ href: (c.forms || {}).resourceFormUrl, label: 'Suggest a resource', primary: true }],
        [{ id: 'sec-resource-filters', label: 'Filter' }, { id: 'resourceList', label: 'Directory' }]) +
      '<div class="filters" id="sec-resource-filters"><label>County <select id="fCounty"><option value="">All</option><option>Cecil</option><option>Harford</option><option value="both">Both / regional</option><option>Statewide</option></select></label>' +
      '<label>Category <select id="fCat"><option value="">All</option>' + cats.map(function (x) { return '<option>' + esc(x) + '</option>'; }).join('') + '</select></label>' +
      '<label>Level <select id="fType"><option value="">All</option>' + types.map(function (x) { return '<option>' + esc(x) + '</option>'; }).join('') + '</select></label>' +
      indicatorSelect('fResInd', all, 'Indicators') +
      '<label>Search <input id="fText" type="search" placeholder="name or keyword"></label></div>' +
      accBar('resourceList', 'fCount') + '<div id="resourceList"></div>';

    function apply() {
      var county = $('#fCounty').value.toLowerCase(), cat = $('#fCat').value,
          type = $('#fType').value, ind = $('#fResInd').value, q = $('#fText').value.toLowerCase();
      var list = all.filter(function (r) {
        if (county && !countyMatches(String(r.Counties || '').toLowerCase(), county)) return false;
        if (cat && r.Category !== cat) return false;
        if (type && r.Type !== type) return false;
        if (!hasInd(r, 'Indicators', ind)) return false;
        if (q && (String(r.Name) + ' ' + String(r.Description) + ' ' + String(r.Eligibility)).toLowerCase().indexOf(q) < 0) return false;
        return true;
      });
      $('#fCount').textContent = list.length + ' of ' + all.length;

      // Group into collapsible categories so 60 programs read as a directory,
      // not a wall. A filtered result set opens automatically.
      var groups = {};
      list.forEach(function (r) { (groups[r.Category || 'Other'] = groups[r.Category || 'Other'] || []).push(r); });
      var names = Object.keys(groups).sort();
      var narrowed = !!(county || cat || type || ind || q);
      $('#resourceList').innerHTML = names.map(function (g) {
        var rows = groups[g].map(function (r) {
          return disc({
            title: esc(r.Name),
            meta: [esc(r.Type), esc(r.Counties)].filter(Boolean).join(' · '),
            tags: indicatorPills(r.Indicators)
          }, para(r.Description) +
            (r.Eligibility ? '<p><strong>Eligibility:</strong> ' + esc(r.Eligibility) + '</p>' : '') +
            (r.Contact ? '<p><strong>Contact:</strong> ' + esc(r.Contact) + '</p>' : '') +
            linksLine(r) + provenanceLine(r));
        }).join('');
        return disc({ title: esc(g), meta: groups[g].length + ' program' + (groups[g].length === 1 ? '' : 's') },
          rows, narrowed || names.length === 1);
      }).join('') || '<p class="muted">No resources match.</p>';
    }
    ['fCounty', 'fCat', 'fType', 'fResInd'].forEach(function (id) { $('#' + id).addEventListener('change', apply); });
    $('#fText').addEventListener('input', apply);
    apply();
  }

  /* ---------- regional data ---------- */
  function renderData(c) {
    var rows = c.data || [];
    function fmt(v, unit) { if (v === '' || v === undefined) return '—'; return unit === '%' ? v + '%' : String(v); }
    function cell(v, md, unit, name) {
      var n = parseFloat(v), m = parseFloat(md);
      var higherBetter = /early prenatal|adequate|coverage|breastfeed/i.test(String(name));
      var cls = '';
      if (!isNaN(n) && !isNaN(m) && n !== m) cls = (higherBetter ? n > m : n < m) ? 'better' : 'worse';
      return '<td class="num ' + cls + '">' + esc(fmt(v, unit)) + '</td>';
    }
    var html = '<h2>Regional Data: the six indicators</h2>' +
      '<p class="lead">These six outcomes are the problems the Task Force exists to improve. Red marks a county doing worse than Maryland, green better.</p>' +
      pageBar([], [{ id: 'sec-data-table', label: 'Indicator table' }]) +
      '<div class="table-scroll" id="sec-data-table"><table><thead><tr><th>Indicator</th><th>Cecil</th><th>Harford</th><th>Maryland</th><th>Year</th><th>Source</th></tr></thead><tbody>';
    rows.forEach(function (r) {
      html += '<tr><td><strong>' + esc(r.Indicator) + '</strong>' +
        (r.Definition ? '<div class="small muted">' + esc(r.Definition) + '</div>' : '') + '</td>' +
        cell(r.Cecil, r.Maryland, r.Unit, r.Indicator) + cell(r.Harford, r.Maryland, r.Unit, r.Indicator) +
        '<td class="num">' + esc(fmt(r.Maryland, r.Unit)) + '</td><td>' + esc(r.Year) + '</td>' +
        '<td class="small">' + (r.SourceURL ? link(r.SourceURL, r.Source) : esc(r.Source)) + '</td></tr>';
    });
    html += '</tbody></table></div>' +
      '<p class="legend">For most indicators lower is better; for early prenatal care, higher is better.</p>' +
      '<p style="margin-top:16px"><a class="btn" href="#evidence" data-goto="evidence">Why these numbers? See Drivers &amp; Evidence</a></p>';
    $('#view-data').innerHTML = html;
  }

  /* ---------- drivers and evidence ---------- */
  function dataCol(label, value, url) {
    if (!value) return '';
    var cite;
    if (url) cite = '<span class="cite">' + link(url, 'Source') + '</span>';
    else if (/not yet obtained|not been obtained|has not been retrieved|not retrieved|unconfirmed/i.test(value))
      cite = '<span class="cite muted">Data gap — on the request list</span>';
    else cite = '<span class="cite muted">Task Force record; no published URL</span>';
    return '<div><strong>' + esc(label) + '</strong>' + esc(value) + cite + '</div>';
  }
  function renderEvidence(c) {
    var drivers = c.drivers || [], cases = c.caseStudies || [];
    var byInd = {};
    drivers.forEach(function (d) { var k = d.Indicator || 'Cross-cutting'; (byInd[k] = byInd[k] || []).push(d); });
    var keys = Object.keys(byInd).sort(function (a, b) {
      var ia = INDICATOR_ORDER.indexOf(a), ib = INDICATOR_ORDER.indexOf(b);
      return (ia < 0 ? 99 : ia) - (ib < 0 ? 99 : ib);
    });

    var html = '<h2>Drivers &amp; Evidence</h2>' +
      '<p class="lead">What the research says causes each indicator, what the national, Maryland and local data show, and where comparable communities moved their numbers. Use this to pick actions, not just name problems.</p>' +
      '<p class="notice small">Every figure links to the page it came from. Figures were read from search-index extracts rather than full-text retrieval, so open the primary source before putting a number in front of an external audience. Anything unverified is marked as a gap rather than smoothed over.</p>';

    html += pageBar([], [
      { id: 'sec-drivers', label: 'What drives each indicator' },
      { id: 'sec-cases', label: 'Communities that moved their numbers' }
    ]);
    html += '<div class="section" id="sec-drivers"><h3>What drives each indicator</h3>' +
      '<div class="filters">' +
      '<label>Indicator <select id="fInd"><option value="">All</option>' + keys.map(function (k) { return '<option>' + esc(k) + '</option>'; }).join('') + '</select></label>' +
      '<label>Action Team <select id="fTeam"><option value="">All</option>' + uniq(drivers.map(function (d) { return d.RelatedActionTeam; })).map(function (k) { return '<option>' + esc(k) + '</option>'; }).join('') + '</select></label>' +
      '</div>' + accBar('driverList', 'dCount') + '<div id="driverList"></div></div>';

    html += '<div class="section" id="sec-cases"><h3>Communities that moved their numbers</h3>' +
      '<p class="muted small">Before and after results as reported by the source. Evidence quality is stated: a pre/post result without a comparison group is weaker than a trial. Four entries are included because their result was null or their gains were unequal.</p>' +
      '<div class="filters"><label>Indicator <select id="fCaseInd"><option value="">All</option>' +
      uniq(cases.reduce(function (a, cs) { return a.concat(splitInd(cs.Indicators)); }, [])).map(function (k) { return '<option>' + esc(k) + '</option>'; }).join('') +
      '</select></label></div>' + accBar('caseList', 'cCount') + '<div id="caseList"></div></div>';
    $('#view-evidence').innerHTML = html;

    function driverDisc(d) {
      var body = para(d.WhatTheEvidenceShows) +
        '<div class="cols">' +
        dataCol('National', d.NationalData, d.NationalSourceURL) +
        dataCol('Maryland', d.MarylandData, d.MarylandSourceURL) +
        dataCol('Cecil / Harford', d.LocalData, d.LocalSourceURL) +
        '</div>' +
        (d.ImplicationForAction ? '<div class="implication"><strong>Implication for action.</strong> ' + esc(d.ImplicationForAction) + '</div>' : '') +
        (d.DataSystem ? '<p class="small muted" style="margin-top:10px"><strong>Where to get county-level data:</strong> ' + esc(d.DataSystem) + '</p>' : '') +
        '<p class="meta">' + (d.SourceURL ? link(d.SourceURL, d.Source || 'Source') : esc(d.Source || '')) + '</p>';
      return disc({
        title: esc(d.Driver),
        meta: truncate(esc(d.ImplicationForAction || d.WhatTheEvidenceShows), 130),
        tags: (d.RelatedActionTeam ? '<span class="pill">' + esc(d.RelatedActionTeam) + '</span>' : '') +
          (d.Confidence ? '<span class="pill ind-tag">Confidence: ' + esc(d.Confidence) + '</span>' : '')
      }, body);
    }
    function caseDisc(cs) {
      var body = '<p><strong>Problem.</strong> ' + esc(cs.ProblemFaced) + '</p>' +
        '<p><strong>What they did.</strong> ' + esc(cs.WhatTheyDid) + '</p>' +
        '<div class="implication"><span class="result">Result.</span> ' + esc(cs.Result) + '</div>' +
        (cs.EvidenceQuality ? '<p class="small"><strong>Evidence quality:</strong> ' + esc(cs.EvidenceQuality) + '</p>' : '') +
        (cs.Funding ? '<p class="small"><strong>Funding:</strong> ' + esc(cs.Funding) + '</p>' : '') +
        (cs.Transferability ? '<p><strong>For Cecil and Harford.</strong> ' + esc(cs.Transferability) + '</p>' : '') +
        '<p class="meta">' + link(cs.SourceURL, 'Source') + '</p>';
      return disc({
        title: esc(cs.Name),
        meta: [esc(cs.Location), esc(cs.Years), esc(cs.EvidenceQuality)].filter(Boolean).join(' · '),
        tags: indicatorPills(cs.Indicators)
      }, body);
    }
    function applyDrivers() {
      var ind = $('#fInd').value, team = $('#fTeam').value;
      var list = drivers.filter(function (d) {
        return (!ind || (d.Indicator || 'Cross-cutting') === ind) && (!team || d.RelatedActionTeam === team);
      });
      $('#dCount').textContent = list.length + ' of ' + drivers.length;
      var out = '';
      keys.forEach(function (k) {
        var group = list.filter(function (d) { return (d.Indicator || 'Cross-cutting') === k; });
        if (!group.length) return;
        out += disc({ title: esc(k), meta: group.length + ' driver' + (group.length === 1 ? '' : 's') },
          group.map(driverDisc).join(''), !!ind);
      });
      $('#driverList').innerHTML = out || '<p class="muted">No drivers match.</p>';
    }
    function applyCases() {
      var ind = $('#fCaseInd').value;
      var list = cases.filter(function (cs) { return !ind || splitInd(cs.Indicators).indexOf(ind) >= 0; });
      $('#cCount').textContent = list.length + ' of ' + cases.length;
      $('#caseList').innerHTML = list.map(caseDisc).join('') || '<p class="muted">No case studies match.</p>';
    }
    $('#fInd').addEventListener('change', applyDrivers);
    $('#fTeam').addEventListener('change', applyDrivers);
    $('#fCaseInd').addEventListener('change', applyCases);
    applyDrivers(); applyCases();
  }

  /* ---------- meetings and actions ---------- */
  function renderMeetings(c) {
    var open = (c.actionItems || []).filter(function (a) { return !/done|closed|complete/i.test(a.Status); });
    var done = (c.actionItems || []).filter(function (a) { return /done|closed|complete/i.test(a.Status); });
    function actionTable(list) {
      if (!list.length) return '<p class="muted small">None.</p>';
      return '<div class="table-scroll"><table><thead><tr><th>Item</th><th>Team</th><th>Indicators</th><th>Owner</th><th>Due</th><th>Status</th></tr></thead><tbody>' +
        list.map(function (a) {
          return '<tr><td>' + esc(a.Item) + '</td><td>' + esc(a.ActionTeam) + '</td><td>' + indicatorPills(a.Indicators) +
            '</td><td>' + esc(a.Owner) + '</td><td>' + esc(a.DueDate) + '</td><td>' + statusPill(a.Status) + '</td></tr>';
        }).join('') + '</tbody></table></div>';
    }
    var html = '<h2>Meetings &amp; Action Items</h2>' +
      pageBar([], [{ id: 'sec-open-actions', label: 'Open action items' }, { id: 'sec-meetings', label: 'Meeting summaries' }]
        .concat(done.length ? [{ id: 'sec-done-actions', label: 'Completed' }] : [])) +
      '<div class="section" id="sec-open-actions"><h3>Open action items (' + open.length + ')</h3>' + actionTable(open) + '</div>' +
      '<div class="section" id="sec-meetings"><h3>Meeting summaries</h3><div id="meetingList">' +
      ((c.meetings || []).map(function (m, i) {
        return disc({
          title: esc(m.Title),
          meta: [esc(m.Date), esc(m.ActionTeam)].filter(Boolean).join(' · ')
        }, para(m.Summary) +
          (m.Decisions ? '<h4>Decisions</h4>' + bullets(m.Decisions) : '') +
          (m.NextSteps ? '<h4>Next steps</h4>' + bullets(m.NextSteps) : '') +
          (m.NotesURL ? '<p>' + link(m.NotesURL, 'Full notes') + '</p>' : ''), i === 0);
      }).join('') || '<p class="muted">No meeting summaries yet.</p>') + '</div></div>';
    if (done.length) {
      html += '<div class="section" id="sec-done-actions">' + disc({ title: 'Completed action items', meta: done.length + ' item' + (done.length === 1 ? '' : 's') }, actionTable(done)) + '</div>';
    }
    $('#view-meetings').innerHTML = html;
  }

  /* ---------- barrier log ---------- */
  function renderBarriers(c) {
    var b = c.barriers || { total: 0, byCategory: {}, byCounty: {}, byMonth: {}, published: [] };
    function bars(map, chrono) {
      var ks = Object.keys(map);
      if (!ks.length) return '<p class="muted small">No reports yet.</p>';
      ks.sort(chrono ? function (a, x) { return a.localeCompare(x); } : function (a, x) { return map[x] - map[a]; });
      var max = Math.max.apply(null, ks.map(function (k) { return map[k]; }));
      return ks.map(function (k) {
        return '<div class="bar"><span class="lbl">' + esc(k) + '</span><span class="track"><span class="fill" style="width:' +
          Math.round(map[k] / max * 100) + '%"></span></span><span class="n">' + map[k] + '</span></div>';
      }).join('');
    }
    var reports = b.reports || [];
    var html = '<h2>Access Barrier Log</h2>' +
      '<p class="lead">A de-identified record of barriers families hit while trying to reach perinatal care, so the Task Force can see patterns and measure progress. Every report is listed below; open one to read it in full. This is not a clinical record and not a referral pathway.</p>' +
      '<p class="notice"><strong>Never enter identifying details.</strong> No names, initials, dates of birth, addresses, exact dates, or anything that could identify a patient or family. Describe the barrier, not the person. Reports are visible to everyone who can open this page as soon as they are submitted, so there is no step between what you type and what is published.</p>' +
      pageBar([{ href: (c.forms || {}).barrierFormUrl, label: 'Report a barrier', primary: true }],
        [{ id: 'sec-barrier-summary', label: 'Summary' }, { id: 'sec-barrier-reports', label: 'Reports' }]) +
      '<div class="grid" id="sec-barrier-summary"><div class="card"><h3>Total reports</h3><p style="font-size:34px;font-weight:700;color:var(--navy-800);margin:0">' + esc(b.total) + '</p></div>' +
      '<div class="card"><h3>By county</h3>' + bars(b.byCounty) + '</div>' +
      '<div class="card"><h3>By month</h3>' + bars(b.byMonth, true) + '</div></div>' +
      '<div class="section" style="margin-top:14px"><div class="card"><h3>By barrier category</h3>' + bars(b.byCategory) + '</div></div>';

    html += '<div class="section" id="sec-barrier-reports"><h3>Reports (' + reports.length + ')</h3>';
    if (!reports.length) {
      html += '<p class="muted">No barriers have been reported yet. The first report will appear here as soon as it is submitted.</p></div>';
    } else {
      html += '<div class="filters">' +
        '<label>County <select id="fBarCounty"><option value="">All</option>' +
        uniq(reports.map(function (r) { return r.county; })).map(function (x) { return '<option>' + esc(x) + '</option>'; }).join('') + '</select></label>' +
        '<label>Category <select id="fBarCat"><option value="">All</option>' +
        uniq(reports.map(function (r) { return r.category; })).map(function (x) { return '<option>' + esc(x) + '</option>'; }).join('') + '</select></label>' +
        '<label>Service <select id="fBarSvc"><option value="">All</option>' +
        uniq(reports.map(function (r) { return r.service; })).map(function (x) { return '<option>' + esc(x) + '</option>'; }).join('') + '</select></label>' +
        '</div>' + accBar('barrierList', 'bCount') + '<div id="barrierList"></div></div>';
    }
    $('#view-barriers').innerHTML = html;

    if (!reports.length) return;
    function applyBarriers() {
      var county = $('#fBarCounty').value, cat = $('#fBarCat').value, svc = $('#fBarSvc').value;
      var list = reports.filter(function (r) {
        return (!county || r.county === county) && (!cat || r.category === cat) && (!svc || r.service === svc);
      });
      $('#bCount').textContent = list.length + ' of ' + reports.length;
      $('#barrierList').innerHTML = list.map(function (r) {
        var body =
          '<dl class="report">' +
          '<dt>Month of occurrence</dt><dd>' + esc(r.month) + '</dd>' +
          '<dt>County</dt><dd>' + esc(r.county) + '</dd>' +
          '<dt>Barrier category</dt><dd>' + esc(r.category) + '</dd>' +
          '<dt>Service type</dt><dd>' + esc(r.service) + '</dd>' +
          (r.org ? '<dt>Reporting organisation</dt><dd>' + esc(r.org) + '</dd>' : '') +
          '</dl>' +
          '<h4>What happened</h4>' +
          (r.textWithheld
            ? '<p class="muted">The written description is held until a co-chair has reviewed it for identifying details.</p>'
            : (r.description ? para(r.description) : '<p class="muted">No description was entered.</p>')) +
          '<p class="meta">' + esc(r.reviewStatus) +
          '. The reporter\'s identity is never published. Contact the co-chairs to follow up on a report.</p>';
        return disc({
          title: esc(r.category),
          meta: [esc(r.month), esc(r.county), esc(r.service)].filter(Boolean).join(' · '),
          tags: r.reviewed
            ? '<span class="pill both">Reviewed</span>'
            : '<span class="pill status-pending">Not yet reviewed</span>'
        }, body);
      }).join('') || '<p class="muted">No reports match.</p>';
    }
    ['fBarCounty', 'fBarCat', 'fBarSvc'].forEach(function (id) { $('#' + id).addEventListener('change', applyBarriers); });
    applyBarriers();
  }

  /* ---------- utils ---------- */
  function uniq(arr) {
    var seen = {};
    return arr.filter(function (x) { x = String(x || ''); if (!x || seen[x]) return false; seen[x] = 1; return true; }).sort();
  }
  function truthy(v) {
    var s = String(v === undefined || v === null ? '' : v).toLowerCase();
    return v === true || s === 'true' || s === 'yes' || s === 'y' || s === '1';
  }

  function onContent(c) {
    STATE.content = c;
    $('#status').hidden = true;
    $('#generatedAt').textContent = 'Content refreshed ' + new Date(c.generatedAt).toLocaleString();
    renderHome(c); renderProblems(c); renderStructure(c); renderFunding(c);
    renderResources(c); renderData(c); renderEvidence(c); renderMeetings(c); renderBarriers(c);
    var initial = (location.hash || '#home').replace('#', '');
    show($$('.tab').some(function (t) { return t.dataset.view === initial; }) ? initial : 'home');
  }
  function onError(err) {
    var s = $('#status');
    s.className = 'status error';
    s.textContent = 'Could not load content. Please refresh, or contact the co-chairs if this continues.';
    console.error(err && err.message ? err.message : err);
  }
  google.script.run.withSuccessHandler(onContent).withFailureHandler(onError).getSiteContent();
})();
