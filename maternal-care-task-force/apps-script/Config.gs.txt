/**
 * Returns a Script Property or throws a clear error if it is missing.
 * @param {string} key
 * @return {string}
 */
function getRequiredProperty_(key) {
  var v = PropertiesService.getScriptProperties().getProperty(key);
  if (!v) {
    throw new Error('Missing Script Property "' + key + '". Run runSetup() in Setup.gs first.');
  }
  return v;
}

/**
 * @return {string} Content spreadsheet ID (service-identity access only).
 */
function getContentSpreadsheetId_() {
  return getRequiredProperty_(CONFIG.PROP_CONTENT_SS_ID);
}

/**
 * @return {string} Audit spreadsheet ID.
 */
function getAuditSpreadsheetId_() {
  return getRequiredProperty_(CONFIG.PROP_AUDIT_SS_ID);
}

/**
 * @return {string[]} Lower-cased admin emails.
 */
function getAdminEmails_() {
  var raw = PropertiesService.getScriptProperties().getProperty(CONFIG.PROP_ADMIN_EMAILS) || '';
  return raw.split(',').map(function (e) { return e.trim().toLowerCase(); }).filter(Boolean);
}
